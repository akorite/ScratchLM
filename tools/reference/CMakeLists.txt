cmake_minimum_required(VERSION 3.16)
project(bonsai_reference LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE)
    message(FATAL_ERROR "LLAMA_SOURCE is required")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(bonsai-reference native.cpp)
target_compile_features(bonsai-reference PRIVATE cxx_std_17)
target_include_directories(bonsai-reference PRIVATE "${LLAMA_SOURCE}/ggml/src")
target_link_libraries(bonsai-reference PRIVATE llama ggml-cpu)
